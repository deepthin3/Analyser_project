             PGM        PARM(&CALLINGMOD &ACTION &PAIRID &PUSRID &PROLE)

             DCL        VAR(&SIGNOUT) TYPE(*DEC) LEN(1 0)
             DCL        VAR(&CALLINGMOD) TYPE(*CHAR) LEN(10)
             DCL        VAR(&ACTION) TYPE(*CHAR) LEN(10)
             DCL        VAR(&PAIRID) TYPE(*CHAR) LEN(5)
             DCL        VAR(&PUSRID) TYPE(*CHAR) LEN(5)
             DCL        VAR(&PFLTID) TYPE(*CHAR) LEN(5)
             DCL        VAR(&PBKID) TYPE(*CHAR) LEN(5)
             DCL        VAR(&Prole) TYPE(*CHAR) LEN(10)
             DCL        VAR(&EXITFLAG) TYPE(*LGL) VALUE('0')
             DCL        VAR(&CDATE) TYPE(*CHAR) LEN(6)
             DCL        VAR(&CTIME) TYPE(*CHAR) LEN(6)
             DCL        VAR(&EXITFLAG) TYPE(*LGL) VALUE('0')
             DCLF       FILE(ADMMNU)
             INCLUDE    SRCMBR(SYSCOPYCL) SRCFILE(*LIBL/QCLSRC)

             MONMSG     MSGID(CPF0000) EXEC(GOTO CMDLBL(ERROR))
             CHGVAR     VAR(&SIGNOUT) VALUE(00)
/* Display menu and route based on choice */
LOOP:
             RTVSYSVAL  SYSVAL(QDATE) RTNVAR(&CDATE)
             RTVSYSVAL  SYSVAL(QTIME) RTNVAR(&CTIME)
             CVTDAT     DATE(&CDATE) TOVAR(&SYSDATE) TOFMT(*YMD) +
                          TOSEP(/)
             CHGVAR     VAR(&DUSER) VALUE(&PUSRID)
             CHGVAR     VAR(&DCHOICE) VALUE(0)
             CHGVAR     VAR(&SYSTIME) VALUE(%SST(&CTIME  1 +
                          2) *CAT ':' *CAT %SST(&CTIME  3 2) +
                          *CAT ':' *cat %SST(&CTIME  5 2))
             SNDRCVF    RCDFMT(ADMNDASHR )
             MONMSG     MSGID(CPF0000) EXEC(GOTO CMDLBL(EXIT))
             CHGVAR     VAR(&CallingMod) VALUE("ADMIN")

             IF         COND(&IN03 *EQ '1' *OR &IN12 *EQ '1') +
                          THEN(GOTO CMDLBL(EXIT))

             IF         COND(&IN05 = '1') THEN(DO)
             CHGVAR     VAR(&DCHOICE)  VALUE(0       )
             ENDDO

             IF         COND(&DCHOICE = 1) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(AIRLINER2) PARM((&CALLINGMOD) (&ACTION) +
                          (&PAIRID) (&PUSRID))
             ENDDO

             IF         COND(&DCHOICE = 2) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(USERRP) PARM((&CALLINGMOD) (&ACTION) +
                          (&PAIRID) (&PUSRID))
             ENDDO

             IF         COND(&DCHOICE = 3) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(BOOKINGRP) PARM((&CALLINGMOD) (&ACTION) +
                          (&PAIRID) (&PUSRID) ('     ') ('     '))
             ENDDO

             IF         COND(&DCHOICE = 4) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(FLIGHTRP) PARM((&CALLINGMOD) (&ACTION) +
                          (&PAIRID) (&PUSRID))
             ENDDO

             IF         COND(&DCHOICE = 5) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(GRIEVERP) PARM((&CALLINGMOD) (&ACTION) +
                          (&PAIRID) (&PUSRID) (&PFLTID) (&PBKID))
             ENDDO

             IF         COND(&DCHOICE = 6) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE('LIST')
             CALL       PGM(CONTROLRP) PARM(('          ') ('          ') +
                          ('          ') (&ACTION) (&CALLINGMOD)+
                          (&PUSRID))
             ENDDO

             IF         COND(&DCHOICE = 7) THEN(DO)
             CHGVAR     VAR(&ACTION) VALUE(UPDATE)
             CALL       PGM(LOGINRP) PARM((&CALLINGMOD) (&ACTION) +
                          (&PUSRID) (&PROLE) (&SIGNOUT))
             ENDDO

             GOTO       CMDLBL(LOOP)

EXIT:
             RETURN

             GOTO       CMDLBL(END)
 ERROR:      SNDPGMMSG  MSGID(CPF9898) MSGF(QCPFMSG) MSGDTA('"An +
                          unexpected error occured."') TOPGMQ(*EXT)
 END:        ENDPGM
